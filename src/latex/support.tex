
\usepackage{xcolor}

\makeatletter
\newcommand{\TexDiffRemovedKey}[1]{%
  \ifcsname r@tex-diff-removed:#1\endcsname
    tex-diff-removed:#1%
  \else
    #1%
  \fi
}

\newcommand{\TexDiffBoxColor}[1]{%
  \let\TexDiffOriginalSetColor\set@color
  \let\set@color\relax
  \color{#1}%
  \let\set@color\TexDiffOriginalSetColor
}
\DeclareRobustCommand{\TexDiffBoxBegin}[1]{%
  \colorlet{TexDiffPreviousColor}{.}%
  \let\TexDiffPreviousNormalColor\normalcolor
  \def\normalcolor{\color{#1}}%
  \TexDiffBoxColor{#1}%
}
\DeclareRobustCommand{\TexDiffBoxEnd}{%
  \let\normalcolor\TexDiffPreviousNormalColor
  \TexDiffBoxColor{TexDiffPreviousColor}%
}
\let\TexDiffParagraphColor\relax
\AddToHook{para/begin}[tex-diff/paragraph]{\TexDiffParagraphColor}
\newcommand{\TexDiffEnvironmentColor}[1]{%
  \def\normalcolor{\color{#1}}%
  \def\TexDiffParagraphColor{\color{#1}}%
  \TexDiffBoxColor{#1}%
}
\newcommand{\TexDiffMathTagColor}[1]{%
  \ifdefined\maketag@@@
    \let\TexDiffOriginalMathTag\maketag@@@
    \def\maketag@@@##1{\TexDiffOriginalMathTag{\color{#1}##1}}%
  \fi
}
\newcommand{\TexDiffAlignmentColor}[1]{%
  \def\normalcolor{\color{#1}}%
  \everymath\expandafter{\the\everymath\color{#1}}%
  \TexDiffMathTagColor{#1}%
}
\makeatother
